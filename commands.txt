# Run these commands from the workspace root containing the BlazorApp1 folder.

# Restore project dependencies
dotnet restore BlazorApp1/BlazorApp1.csproj

# Build the project
dotnet build BlazorApp1/BlazorApp1.csproj

# Run using the HTTP profile, then open http://127.0.0.1:5236
dotnet run --project BlazorApp1/BlazorApp1.csproj --launch-profile http

# Run using the HTTPS profile, then open https://localhost:7232
dotnet run --project BlazorApp1/BlazorApp1.csproj --launch-profile https

# Run with hot reload while editing
dotnet watch --project BlazorApp1/BlazorApp1.csproj --launch-profile http

# Remove generated build output
dotnet clean BlazorApp1/BlazorApp1.csproj

# Publish a Release build to BlazorApp1/bin/Release/net10.0/publish
dotnet publish BlazorApp1/BlazorApp1.csproj --configuration Release
